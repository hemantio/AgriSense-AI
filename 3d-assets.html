Three.js Organic Flow Controller

const initParticles = (scene) => { const particleCount = 2000; const geometry = new THREE.BufferGeometry(); const positions = new Float32Array(particleCount * 3); const velocities = []; for (let i = 0; i < particleCount; i++) { positions[i * 3] = (Math.random() - 0.5) * 50; positions[i * 3 + 1] = Math.random() * 30; positions[i * 3 + 2] = (Math.random() - 0.5) * 50; velocities.push({ x: -0.05 - Math.random() * 0.05, y: -0.15 - Math.random() * 0.1, z: (Math.random() - 0.5) * 0.02 }); } geometry.setAttribute('position', new THREE.BufferAttribute(positions, 3)); const material = new THREE.PointsMaterial({ color: 0x10b981, size: 0.15, transparent: true, opacity: 0.6, blending: THREE.AdditiveBlending }); const particleSystem = new THREE.Points(geometry, material); scene.add(particleSystem); const updateParticles = () => { const pos = geometry.attributes.position.array; for (let i = 0; i < particleCount; i++) { pos[i * 3] += velocities[i].x; pos[i * 3 + 1] += velocities[i].y; pos[i * 3 + 2] += velocities[i].z; if (pos[i * 3 + 1] < 0) { pos[i * 3] = (Math.random() - 0.5) * 50; pos[i * 3 + 1] = 30; pos[i * 3 + 2] = (Math.random() - 0.5) * 50; } } geometry.attributes.position.needsUpdate = true; }; return { system: particleSystem, update: updateParticles }; };
